window.onerror = function(message, source, lineno, colno, error) {
    var errorText = 'Ошибка: ' + message + '\n' +
                    'Файл: ' + source + '\n' +
                    'Строка: ' + lineno + ', колонка: ' + colno + '\n' +
                   'Стек: ' + (error && error.stack ? error.stack : 'нет данных');
    alert(errorText);
    return false;
};

// Функция для перемещения кнопки "Show password" внутрь поля ввода
function movePasswordToggleInside() {
    var passwordInput = document.getElementById('password');
    var passwordToggle = document.querySelector('[data-password-toggle]');
    var passwordContainer = document.querySelector('.password-container');
    
    if (!passwordInput || !passwordToggle) return;
    
    if (!passwordContainer) {
        passwordContainer = document.createElement('div');
        passwordContainer.className = 'password-container';
        passwordContainer.style.position = 'relative';
        passwordContainer.style.width = '100%';
        passwordInput.parentNode.insertBefore(passwordContainer, passwordInput);
        passwordContainer.appendChild(passwordInput);
    }
    
    passwordContainer.appendChild(passwordToggle);
    
    passwordToggle.style.position = 'absolute';
    passwordToggle.style.right = '8px';
    passwordToggle.style.top = '50%';
    passwordToggle.style.transform = 'translateY(-50%)';
    passwordToggle.style.background = 'transparent';
    passwordToggle.style.border = 'none';
    passwordToggle.style.padding = '0';
    passwordToggle.style.cursor = 'pointer';
    
    passwordInput.style.paddingRight = '30px';
}

}

function StartFormat() {
    var usernameInput = document.getElementById('username');
    var loginForm = document.getElementById('kc-form-login');
    var submitButton = document.getElementById('kc-login');
    
    if (!usernameInput || !loginForm || !submitButton) return;
    
    usernameInput.placeholder = 'Email или +7 (XXX) XXX-XX-XX';
    
    usernameInput.addEventListener('input', InputProcess);
    loginForm.addEventListener('submit', FormSubmitProcess);
}

StartFormat();

function FormSubmitProcess(e) {
    var usernameInput = document.getElementById('username');
    var submitButton = document.getElementById('kc-login');
    
    if (!usernameInput || !submitButton) return;
    
    if (usernameInput.dataset && usernameInput.dataset.rawValue) {
        usernameInput.value = usernameInput.dataset.rawValue;
    }
    
    submitButton.disabled = true;
    
}

function InputProcess(e) {
    var value = this.value;
    var looksLikePhone = value.indexOf('+7') === 0 || /^[\d+()\- ]*$/.test(value);
    var looksLikeEmail = value.indexOf('@') !== -1 || /^[^@]*@[^@]*$/.test(value);
    
    // Определяем, была ли нажата Backspace или Del
    var isBackspacePressed = e.inputType === 'deleteContentBackward' || 
                           e.inputType === 'deleteContentForward' || 
                           e.key === 'Backspace' || 
                           e.key === 'Delete';
    
    if (looksLikePhone && !looksLikeEmail) {
        formatAsPhone(e.target, isBackspacePressed);
    }
}

function formatAsPhone(input, isBackspacePressed = false) {
    var cursorPosition = input.selectionStart;
    var value = input.value;
    
    // Если пытаемся удалить +7 (Backspace или Del)
    if (isBackspacePressed && (value === '+7' || value === '+7 ' || value === '+7 (')) {
        input.value = '';
        input.dataset.rawValue = '';
        return;
    }
    
    var cleaned = value.replace(/\D/g, '');
    
    // Если ничего не введено после +7
    if (cleaned === '7' && (cursorPosition <= 2 || isBackspacePressed)) {
        input.value = '';
        input.dataset.rawValue = '';
        return;
    }
    
    // Стандартное форматирование
    if (cleaned.length > 0 && !cleaned.startsWith('7')) {
        cleaned = '7' + cleaned;
    }
    
    var formatted = '+7';
    if (cleaned.length > 1) formatted += ' (' + cleaned.substring(1, 4);
    if (cleaned.length > 4) formatted += ') ' + cleaned.substring(4, 7);
    if (cleaned.length > 7) formatted += '-' + cleaned.substring(7, 9);
    if (cleaned.length > 9) formatted += '-' + cleaned.substring(9, 11);
    
    input.value = formatted;
    input.dataset.rawValue = cleaned;
    

}
